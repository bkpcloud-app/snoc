   }
        foreach ($file in @(Get-ChildItem -Path $IncludeDir -Filter "*.bkp-*" -ErrorAction SilentlyContinue | Where-Object { -not $_.PSIsContainer })) {
            $result += New-Object PSObject -Property @{
                Source=$file.FullName
                Relative=("zabbix_agentd.d\\" + $file.Name)
                Kind="ARQUIVO"
            }
        }
    }

    if (Test-Path $ScriptsDir) {
        foreach ($file in @(Get-ChildItem -Path $ScriptsDir -Filter "*.bkp-*" -ErrorAction SilentlyContinue | Where-Object { -not $_.PSIsContainer })) {
            $result += New-Object PSObject -Property @{
                Source=$file.FullName
                Relative=("scripts\\" + $file.Name)
                Kind="ARQUIVO"
            }
        }
    }

    if (Test-Path $InstallDir) {
        foreach ($file in @(Get-ChildItem -Path $InstallDir -Filter "zabbix_agentd.conf.bkp-*" -ErrorAction SilentlyContinue | Where-Object { -not $_.PSIsContainer })) {
            $result += New-Object PSObject -Property @{
                Source=$file.FullName
                Relative=$file.Name
                Kind="ARQUIVO"
            }
        }
    }

    return @($result | Sort-Object Relative -Unique)
}

function Archive-LegacyBackupArtifacts {
    param([object[]]$Artifacts, [string]$ArchiveDir)
    if ($null -eq $Artifacts -or $Artifacts.Count -eq 0) { return }
    foreach ($artifact in $Artifacts) {
        if (-not (Test-Path $artifact.Source)) { continue }
        $target = Join-Path $ArchiveDir $artifact.Relative
        $targetParent = Split-Path -Parent $target
        if (-not (Test-Path $targetParent)) { New-Item -Path $targetParent -ItemType Directory -Force | Out-Null }
        Move-Item -Path $artifact.Source -Destination $target -Force
        Write-Log ("Backup legado arquivado: {0}" -f $artifact.Relative)
    }
}

function Test-LiveTreeClean {
    param([object[]]$DesiredFiles)

    $remaining = @(Get-LegacyBackupArtifacts)
    if ($remaining.Count -gt 0) {
        throw "Ainda existem backups legados nas pastas ativas: $($remaining.Relative -join ', ')"
    }

    foreach ($entry in $DesiredFiles) {
        if (-not (Test-Path $entry.Destination)) {
            throw "Arquivo obrigatorio nao foi instalado: $($entry.Relative)"
        }
        if (Test-FileDifferent -Source $entry.Source -Destination $entry.Destination) {
            throw "Arquivo instalado diverge do pacote: $($entry.Relative)"
        }
    }
}

function Backup-File {
    param([string]$Path, [string]$Relative, [string]$BackupDir)
    if (-not (Test-Path $Path)) { return }
    $target = Join-Path $BackupDir $Relative
    $targetDir = Split-Path -Parent $target
    if (-not (Test-Path $targetDir)) { New-Item -Path $targetDir -ItemType Directory -Force | Out-Null }
    Copy-Item -Path $Path -Destination $target -Force
}

function Remove-OldAgentMsiProducts {
    $paths = @(
        "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\*",
        "HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*"
    )
    $oldApps = @(Get-ItemProperty -Path $paths -ErrorAction SilentlyContinue |
        Where-Object { $_.DisplayName -like "Zabbix Agent*" -and $_.DisplayName -notlike "Zabbix Agent 2*" })
    foreach ($app in $oldApps) {
        $productCode = [string]$app.PSChildName
        if ($productCode -match '^\{[0-9A-Fa-f-]+\}$') {
            $uninstallLog = Join-Path $env:TEMP ("zabbix-agent-uninstall-{0}-{1}.log" -f $env:COMPUTERNAME, $RunId)
            Write-Log "Removendo registro MSI antigo: $($app.DisplayName) $($app.DisplayVersion)" "WARN"
            $process = Start-Process -FilePath "msiexec.exe" -ArgumentList @(
                "/x", $productCode, "/qn", "/norestart", "/L*v", "`"$uninstallLog`""
            ) -Wait -PassThru
            if ($process.ExitCode -ne 0 -and $process.ExitCode -ne 1605 -and $process.ExitCode -ne 3010 -and $process.ExitCode -ne 1641) {
                throw "Falha ao remover MSI antigo. ExitCode=$($process.ExitCode). Log=$uninstallLog"
            }
        }
    }
}

function Restore-Backup {
    param([string]$BackupDir, [string[]]$CreatedPaths)
    if ($CreatedPaths) {
        foreach ($created in $CreatedPaths) {
            if (Test-Path $created) { Remove-Item -Path $created -Force -ErrorAction SilentlyContinue }
        }
    }
    if ((Test-IsBlank $BackupDir) -or -not (Test-Path $BackupDir)) { return }
    foreach ($file in @(Get-ChildItem -Path $BackupDir -Recurse -ErrorAction SilentlyContinue | Where-Object { -not $_.PSIsContainer })) {
        $relative = $file.FullName.Substring($BackupDir.Length).TrimStart([char]92)
        if ($relative -eq "zabbix_agentd.conf") { $destination = $ConfPath }
        else { $destination = Join-Path $InstallDir $relative }
        $destinationDir = Split-Path -Parent $destination
        if (-not (Test-Path $destinationDir)) { New-Item -Path $destinationDir -ItemType Directory -Force | Out-Null }
        Copy-Item -Path $file.FullName -Destination $destination -Force
    }
}
