 "`"$MsiPath`"", "/qn", "/norestart", "DONOTSTART=1",
        "SERVER=$Proxy", "SERVERACTIVE=$Proxy", "HOSTNAME=$Hostname", "HOSTMETADATA=$Metadata",
        "LISTENPORT=$($ProductConfig.ListenPort)", "STARTAGENTS=$($ProductConfig.StartAgents)",
        "/L*v", "`"$msiLog`""
    )
    Write-Log "Executando MSI. Log: $msiLog"
    $process = Start-Process -FilePath "msiexec.exe" -ArgumentList $arguments -Wait -PassThru
    if ($process.ExitCode -eq 1603) {
        $logText = ""
        try { if (Test-Path $msiLog) { $logText = [System.IO.File]::ReadAllText($msiLog) } } catch {}
        if ($logText -match "clear the previous agent registration") {
            Write-Log "MSI solicitou limpeza do registro anterior. Limpando e repetindo uma vez." "WARN"
            Clear-ZabbixServiceRegistration
            Remove-OldAgentMsiProducts
            $retryLog = Join-Path $env:TEMP ("zabbix-agent-retry-{0}-{1}.log" -f $env:COMPUTERNAME, $RunId)
            $arguments[$arguments.Count - 1] = "`"$retryLog`""
            $process = Start-Process -FilePath "msiexec.exe" -ArgumentList $arguments -Wait -PassThru
            $msiLog = $retryLog
        }
    }
    if ($process.ExitCode -ne 0 -and $process.ExitCode -ne 3010 -and $process.ExitCode -ne 1641) {
        throw "Falha no MSI. ExitCode=$($process.ExitCode). Log=$msiLog"
    }
    Write-Log "MSI concluido. ExitCode=$($process.ExitCode)"
}

function Remove-OldLogsAndBackups {
    try {
        Get-ChildItem $LogDir -ErrorAction SilentlyContinue | Where-Object { -not $_.PSIsContainer -and $_.LastWriteTime -lt (Get-Date).AddDays(-[int]$ProductConfig.LogRetentionDays) } | Remove-Item -Force -ErrorAction SilentlyContinue
        Get-ChildItem $BackupRoot -ErrorAction SilentlyContinue | Where-Object { $_.PSIsContainer -and $_.LastWriteTime -lt (Get-Date).AddDays(-[int]$ProductConfig.BackupRetentionDays) } | Remove-Item -Recurse -Force -ErrorAction SilentlyContinue
        Get-ChildItem $LegacyArchiveRoot -ErrorAction SilentlyContinue | Where-Object { $_.PSIsContainer -and $_.LastWriteTime -lt (Get-Date).AddDays(-[int]$ProductConfig.LegacyBackupRetentionDays) } | Remove-Item -Recurse -Force -ErrorAction SilentlyContinue
    }
    catch { Write-Log "Falha nao critica na limpeza de logs/backups: $($_.Exception.Message)" "WARN" }
}
